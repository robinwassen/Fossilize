import concurrent.futures,json,subprocess,threading
from pathlib import Path
r=Path('/tmp/fossilize-56-refresh'); o=r/'v2-impl-output'; w=Path('/home/robin/dev/fossilize-issue-56'); intents=json.load(open(r/'v2-impl-intents.json')); rows=[]; mutex=threading.Lock()
def run(item):
 argv=['python3','-B',str(w/'.agents/skills/implement-change-harness/scripts/verification_lock.py'),'--repo',str(w),'--timeout','600','--intent-id',item['intent_id'],'--receipt',item['receipt'],'--']+item['child_command']
 assert not Path(item['receipt']).exists()
 with open(item['log'],'w') as log: code=subprocess.run(argv,cwd=w,stdout=log,stderr=subprocess.STDOUT).returncode
 rec=json.load(open(item['receipt'])); expected=8 if item['intent_id'] in ['v2i-baseline-sigmask','v2i-baseline-stdout-dup2','v2i-baseline-stdin-dup2'] else 0

  rows.append(row); (o/'commands.json').write_text(json.dumps(rows,indent=2)+'\n'); print(item['intent_id'],code,'timeout',rec['timed_out'],flush=True)
 assert code==expected and rec['exit_code']==code and not rec['timed_out'],item['intent_id']
def chain(items):
 for item in items: run(item)
groups=[[i for i in intents if i['intent_id'] in ['v2i-source-audit','v2i-changelog-audit']]]

errors=[]

  except Exception as e: errors.append(str(e))
