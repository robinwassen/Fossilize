import importlib.util, json, pathlib, subprocess, sys
repo = pathlib.Path('/home/robin/dev/fossilize-harness-FOS-000025')
base = 'bbe8ae4acd9ad527658bb58e6b06474924a06284'
previous = '7ec4ddcaa019e8d7601aa7dba12d8a642254def8'
def command(args, cwd=repo):
    result = subprocess.run(args, cwd=cwd, capture_output=True, text=True, timeout=600)
    print(json.dumps({'command': args, 'cwd': str(cwd), 'exit_code': result.returncode}))
    print(result.stdout + result.stderr)
    assert result.returncode == 0
    return result.stdout
candidate = command(['git', 'rev-parse', 'HEAD']).strip()
assert not command(['git', 'status', '--porcelain=v1']).strip()
command(['git', 'diff', '--check', base + '...' + candidate])
command(['git', 'diff', '--check', previous + '...' + candidate])

all_changes = command(['git', 'diff', '--name-status', base + '...' + candidate])
command(['git', 'ls-tree', '-r', '--name-only', 'harness-artifacts'])
command(['git', 'rev-parse', 'HEAD'], pathlib.Path('/home/robin/dev/fossilize'))

sys.path.insert(0, str(repo / '.agents/skills/implement-change-harness/scripts'))
