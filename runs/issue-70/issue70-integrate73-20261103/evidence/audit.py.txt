def require(condition, name):
    if not condition:
        raise AssertionError(name)
    checks.append(name)

def run(argv):
    r = subprocess.run(argv, cwd=SRC, capture_output=True, text=True, timeout=60)
    if r.returncode:
        raise AssertionError((argv, r.returncode, r.stderr))
    return r.stdout

def git(*args):
    return run(['git', *args])

def show(ref, path):
    return git('show', ref + ':' + path)

def patch_lines(a, b, path):
    return [line for line in git('diff', '--unified=0', a, b, '--', path).splitlines()
            if line[:1] in ('+', '-') and not line.startswith(('+++', '---'))]

def assertions(text):
    result = []
    for match in re.finditer(r'checks\.(?:require|contains)\(', text):
        begin = match.start()
        pos = match.end()
        depth = 1
        quote = None
        escape = False
        while depth:
            ch = text[pos]
            pos += 1
            if quote:
                if escape:
                    escape = False
                elif ch == '\\':
                    escape = True
                elif ch == quote:
                    quote = None
            elif ch in ('"', "'"):
                quote = ch
            elif ch == '(':
                depth += 1
            elif ch == ')':
                depth -= 1
        result.append(re.sub(r'\s+', '', text[begin:pos]))
    return collections.Counter(result)

            'passing-complete-log:' + pathlib.Path(log).name)
    return collections.Counter(re.findall(r'Test\s+#\d+:\s+(\S+)\s+\.+\s+Passed', text))
