import collections,json,pathlib,re,subprocess
source=pathlib.Path('/home/robin/.codex/worktrees/4f30/fossilize')
out=pathlib.Path('/tmp/fossilize-issue70-integrate73/output/implementer-3')
subprocess.run(['git','diff','--check'],check=True)
assert not subprocess.check_output(['git','diff','--','cli','replay_event.hpp','test/CMakeLists.txt'],text=True)
base=subprocess.check_output(['git','show','b1e0c5d4f47a2191e0860eeaeea979cc89a8cfa2:CHANGELOG.md'],text=True)
old18=next(l for l in base.splitlines() if 'https://github.com/robinwassen/Fossilize/issues/18' in l)
assert old18 in (source/'CHANGELOG.md').read_text()
old_fixture=subprocess.check_output(['git','show','b1e0c5d4f47a2191e0860eeaeea979cc89a8cfa2:test/fork_child_setup_test.cpp'],text=True)
fixture=(source/'test/fork_child_setup_test.cpp').read_text()
for line in old_fixture.splitlines():
 if 'checks.require' in line: assert line in fixture,('removed assertion line',line)
regex=re.compile(r'^(replay-timeout-|replay-event-|fork-child-|replay-controller-(signal|held-sibling|recover|cli-compatibility)$|replay-outcome-(worker-queue|matrix|creation|cli-compatibility)$|diagnostic-compatibility-test$)')
summary={}
for config in ['debug','release']:
 inventory=json.loads((out/f'impl3-{config}-discovery.log').read_text())
 names=[t['name'] for t in inventory['tests']]
 selected=[n for n in names if regex.search(n)]
 old=json.loads(pathlib.Path(f'/tmp/fossilize-issue70-integrate73/output/implementer-2/impl2-{config}-discovery.json').read_text())
 assert set(names)=={t['name'] for t in old['tests']}
 assert len(names)==260 and len(selected)==99
 results={}
 for gate,expected,count in [('tests',selected,3),('fork-stress',[n for n in names if re.search(r'^fork-child-recovery-(sigmask|stdout-dup2|stdin-dup2)$',n)],25)]:
  log=(out/f'impl3-{config}-{gate}.log').read_text()
  starts=collections.Counter(re.findall(r'Start\s+\d+: ([^\n]+)',log))
  assert starts==collections.Counter({n:count for n in expected}),(config,gate,starts)
  assert '100% tests passed' in log and '***Failed' not in log
  results[gate]={'executed_ids':expected,'repetitions':count,'executions':sum(starts.values()),'status':'passed'}
 summary[config]={'discovered':names,'selected':selected,'discovered_count':len(names),'selected_count':len(selected),'historical_id_union_unchanged':True,'results':results}
 (out/f'impl3-{config}-inventory.json').write_text(json.dumps(summary[config],indent=2)+'\n')
print(json.dumps({'all_original_fork_assertions_preserved':True,'production_and_registrations_unchanged':True,'incoming18_changelog_preserved':True,'variants':{k:{'discovered':v['discovered_count'],'selected':v['selected_count'],'executions':{g:r['executions'] for g,r in v['results'].items()}} for k,v in summary.items()}},indent=2))
