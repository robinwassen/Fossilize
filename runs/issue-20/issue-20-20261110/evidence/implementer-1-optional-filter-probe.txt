file(MAKE_DIRECTORY "${WORK}")
execute_process(COMMAND "${GENERATOR}" --fixture good "${WORK}/valid.foz" RESULT_VARIABLE status TIMEOUT 20)
if(NOT status STREQUAL "0")
    message(FATAL_ERROR "M20/filter-probe-fixture")
endif()
file(REMOVE "${WORK}/absent.foz")
foreach(case healthy-source optional-alone optional-filter)
    set(args --null-device --num-threads 1)
    if(case STREQUAL "healthy-source")
        list(APPEND args --on-disk-replay-whitelist "${WORK}/valid.foz" "${WORK}/valid.foz")
    else()
        list(APPEND args --optional-database-index 0)
        if(case STREQUAL "optional-filter")
            list(APPEND args --on-disk-replay-whitelist "${WORK}/valid.foz")
        endif()
        list(APPEND args "${WORK}/absent.foz")
    endif()
    execute_process(COMMAND "${REPLAY}" ${args} RESULT_VARIABLE status OUTPUT_VARIABLE out ERROR_VARIABLE err TIMEOUT 20)
    if(NOT status STREQUAL "0")
        message(FATAL_ERROR "M20/optional-source-filter-policy/${case}: expected0 got${status}\n${out}\n${err}")
    endif()
endforeach()
