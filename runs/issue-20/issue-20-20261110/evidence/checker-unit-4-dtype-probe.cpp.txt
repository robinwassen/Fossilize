int main()
{
 const char *dir="/tmp/fossilize-h20/output/checker-unit-4/dtype";
 if(mkdir(dir,0700)!=0)return 2;
 {std::unique_ptr<DatabaseInterface> db(create_dumb_folder_database(dir,DatabaseMode::OverWrite));if(!db->prepare()||!db->write_entry(RESOURCE_COMPUTE_PIPELINE,0x20,"{}",2,0))return 2;}
 int fd=socket(AF_UNIX,SOCK_STREAM,0);if(fd<0)return 2;
 sockaddr_un address={};address.sun_family=AF_UNIX;

  if(!exact){printf("FAIL D20/dtype-only-regular-files\n");passed=false;}
 }
 unknown_type=false;unlink(address.sun_path);close(fd);
 return passed?0:1;
}
