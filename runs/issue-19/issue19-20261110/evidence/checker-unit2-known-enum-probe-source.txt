#include "volk.h"
#include "fossilize_feature_filter.hpp"
#include <cstdio>
#include <cstring>
int main() {
 Fossilize::FeatureFilter filter;
 VkPhysicalDeviceFeatures2 f={VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_FEATURES_2};
 VkPhysicalDeviceProperties2 p={VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_PROPERTIES_2};
 if(!filter.init(VK_API_VERSION_1_3,nullptr,0,&f,&p)) return 2;
 VkGraphicsPipelineCreateInfo g={VK_STRUCTURE_TYPE_GRAPHICS_PIPELINE_CREATE_INFO};
 g.renderPass=(VkRenderPass)1;
 Fossilize::SupportDetail d;
 if(!filter.graphics_pipeline_is_supported(&g,d)) return 3;
 VkDynamicState state=VkDynamicState(0x6ffffffd);
 VkPipelineDynamicStateCreateInfo dynamic={VK_STRUCTURE_TYPE_PIPELINE_DYNAMIC_STATE_CREATE_INFO};
 dynamic.dynamicStateCount=1; dynamic.pDynamicStates=&state;g.pDynamicState=&dynamic;
 bool ok=filter.graphics_pipeline_is_supported(&g,d);
 printf("dynamic ok=%d kind=%llu identifier=%s known=%llu raw=%llu expected_raw=%llu\n",ok,(unsigned long long)d.kind,d.identifier,(unsigned long long)d.known,(unsigned long long)d.raw,(unsigned long long)state);
 bool dynamic_correct=!ok && d.kind==Fossilize::SupportKind::Constraint && (d.known&Fossilize::SupportDetail::Raw) && d.raw==uint64_t(state);
 g.pDynamicState=nullptr;
 VkPipelineShaderStageCreateInfo stage={VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO};stage.stage=VK_SHADER_STAGE_COMPUTE_BIT;stage.pName="main";
 g.stageCount=1;g.pStages=&stage;ok=filter.graphics_pipeline_is_supported(&g,d);
 printf("stage ok=%d kind=%llu identifier=%s known=%llu raw=%llu expected_raw=%llu\n",ok,(unsigned long long)d.kind,d.identifier,(unsigned long long)d.known,(unsigned long long)d.raw,(unsigned long long)stage.stage);
 bool stage_correct=!ok && d.kind==Fossilize::SupportKind::Constraint && (d.known&Fossilize::SupportDetail::Raw) && d.raw==uint64_t(stage.stage);
 g.stageCount=0;if(!filter.graphics_pipeline_is_supported(&g,d)||d.kind!=Fossilize::SupportKind::None)return 4;
 VkPipelineLayoutCreateInfo layout={VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};layout.flags=VkPipelineLayoutCreateFlags(0x40000000u);
 bool layout_ok=filter.pipeline_layout_is_supported(&layout,d);
 printf("layout ok=%d kind=%llu identifier=%s known=%llu required=%llu\n",layout_ok,(unsigned long long)d.kind,d.identifier,(unsigned long long)d.known,(unsigned long long)d.required);
 bool layout_correct=!layout_ok && !strcmp(d.identifier,"VkPipelineLayoutCreateInfo.flags");
 layout.flags=0;if(!filter.pipeline_layout_is_supported(&layout,d))return 5;
 if(!dynamic_correct||!stage_correct||!layout_correct){puts("CHECKER19/known-enum-detector-detail assertion failed; supported before/after controls passed");return 1;}
 return 0;
}
